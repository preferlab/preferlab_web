let destinations = [];


Object.entries(regions).forEach(
    ([region, names]) => {

        names.forEach(name => {

            const info =
                travelInfo[name] || null;

            const base =
                regionBase[region];


            let distance =
                base.distance;

            let time =
                base.time;


            /*
               지역 내 행정구역별 차이를
               조금씩 주어 모든 결과가 동일하지 않게 함
            */

            const variation =
                Math.floor(
                    Math.random() * 35
                ) - 15;


            if (
                region !== "서울" &&
                region !== "제주"
            ) {

                distance =
                    Math.max(
                        5,
                        distance + variation
                    );

            }


            let timeText;


            if (region === "제주") {

                timeText =
                    info?.t ||
                    "항공 약 1시간 10분";

            } else if (region === "서울") {

                timeText =
                    "약 20~50분";

            } else {

                const hours =
                    Math.floor(time);

                const minutes =
                    Math.round(
                        (time - hours) * 60
                    );

                const adjustedMinutes =
                    Math.max(
                        0,
                        minutes +
                        Math.floor(
                            Math.random() * 25
                        ) - 10
                    );


                if (hours >= 1) {

                    timeText =
                        `약 ${hours}시간 ${
                            adjustedMinutes
                        }분`;

                } else {

                    timeText =
                        `약 ${adjustedMinutes}분`;

                }

            }


            destinations.push({

                name: name,

                region: region,

                distance:
                    info?.d ||
                    `약 ${distance}km`,

                time:
                    timeText,

                population:
                    info?.p ||
                    "인구 변동 지역",

                description:
                    info?.i ||
                    `${name}은 ${region} 지역의 대표적인 행정구역 중 하나입니다. 지역의 자연과 역사, 생활문화를 경험할 수 있는 여행지입니다.`,

                places:
                    info?.a ||
                    [
                        `${name} 지역 명소`,
                        `${name} 대표 관광지`,
                        `${name} 자연·문화 명소`
                    ],

                foods:
                    info?.f ||
                    [
                        `${name} 향토음식`,
                        `${name} 지역 먹거리`,
                        `${name} 대표 음식`
                    ]

            });

        });

    });


/* ==========================================================

function updateFilterUI() {

    if (selectedRegion === "all") {

        count =
            destinations.length;

        document
            .getElementById("selectedRegion")
            .textContent =
            "전국에서 랜덤 선택";

    } else {

        count =
            destinations.filter(
                city =>
                    city.region ===
                    selectedRegion
            ).length;


        document
            .getElementById("selectedRegion")
            .textContent =
            `${selectedRegion}에서 랜덤 선택`;

    }


    document
        .getElementById("filterCount")
        .textContent =
        `${count}곳`;


    document
        .getElementById("mapCount")
        .textContent =
        `${count}개 여행지`;

}

/* ==========================================================
   RANDOM
========================================================== */

function startRandom() {

    if (isRolling) {
        return;
    }


    isRolling = true;


    const randomBtn =
        document.getElementById(
            "randomBtn"
        );


    const result =
        document.getElementById(
            "result"
        );


    const rollingArea =
        document.getElementById(
            "rollingArea"
        );


    const rollingText =
        document.getElementById(
            "rollingText"
        );


    let candidates;


    if (selectedRegion === "all") {

        candidates =
            destinations;

    } else {

        candidates =
            destinations.filter(
                city =>
                    city.region ===
                    selectedRegion
            );

    }


    if (!candidates.length) {

        isRolling = false;

        alert(
            "선택 가능한 여행지가 없습니다."
        );

        return;

    }


    result.classList.remove("show");

    rollingArea.classList.add("show");


    randomBtn.disabled = true;

    randomBtn.style.opacity = ".5";


    const start =
        Date.now();

    const duration =
        2800;


    const interval =
        setInterval(() => {

            const elapsed =
                Date.now() - start;


            const randomCity =
                candidates[
                    Math.floor(
                        Math.random() *
                        candidates.length
                    )
                ];


            rollingText.textContent =
                randomCity.name;


            if (elapsed >= duration) {

                clearInterval(interval);

                /*
                   마지막 결과는 다시 한번
                   후보 전체에서 독립적으로 추첨
                */

                const finalCity =
                    candidates[
                        Math.floor(
                            Math.random() *
                            candidates.length
                        )
                    ];


                showResult(finalCity);

            }

        }, 80);

}

function showResult(city) {

    document
        .getElementById(
            "rollingArea"
        )
        .classList.remove("show");


    document
        .getElementById(
            "cityName"
        )
        .textContent =
        city.name;


    document
        .getElementById(
            "distance"
        )
        .textContent =
        city.distance;


    document
        .getElementById(
            "travelTime"
        )
        .textContent =
        city.time;


    document
        .getElementById(
            "descriptionText"
        )
        .textContent =
        city.description;


    document
        .getElementById(
            "population"
        )
        .textContent =
        `👥 인구: ${city.population}`;


    const places =
        document.getElementById(
            "places"
        );

    places.innerHTML = "";


    city.places.forEach(
        place => {

            const li =
                document.createElement(
                    "li"
                );

            li.textContent =
                place;

            places.appendChild(li);

        }
    );


    const foods =
        document.getElementById(
            "foods"
        );

    foods.innerHTML = "";


    city.foods.forEach(
        food => {

            const li =
                document.createElement(
                    "li"
                );

            li.textContent =
                food;

            foods.appendChild(li);

        }
    );


    document
        .getElementById(
            "result"
        )
        .classList.add("show");


    const randomBtn =
        document.getElementById(
            "randomBtn"
        );


    randomBtn.disabled = false;

    randomBtn.style.opacity = "1";


    isRolling = false;


    setTimeout(() => {

        document
            .getElementById(
                "result"
            )
            .scrollIntoView({
                behavior: "smooth",
                block: "start"
            });

    }, 100);

}
